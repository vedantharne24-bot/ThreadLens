# Real pthread test programs. Built with debug info so that ThreadLens can
# map lock addresses and call sites back to symbols and source lines.
set(SCENARIO_FLAGS -g -O1 -fno-omit-frame-pointer -Wall -Wextra)

function(add_scenario name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE ${PROJECT_SOURCE_DIR}/collector/include)
    target_compile_options(${name} PRIVATE ${SCENARIO_FLAGS})
    target_link_libraries(${name} PRIVATE Threads::Threads)
    set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
    if(THREADLENS_BUILD_TSAN)
        # ThreadSanitizer twin, used by scripts/evaluate.py for the tool comparison
        add_executable(${name}_tsan ${ARGN})
        target_include_directories(${name}_tsan PRIVATE ${PROJECT_SOURCE_DIR}/collector/include)
        target_compile_options(${name}_tsan PRIVATE -g -O1 -fsanitize=thread)
        target_link_options(${name}_tsan PRIVATE -fsanitize=thread)
        target_link_libraries(${name}_tsan PRIVATE Threads::Threads)
        set_target_properties(${name}_tsan PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin/tsan)
    endif()
endfunction()
